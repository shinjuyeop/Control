cmake_minimum_required(VERSION 3.8)
project(control_tools)
if(POLICY CMP0148)
  cmake_policy(SET CMP0148 OLD)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(kaiev26_msgs REQUIRED)
find_package(lateral_ctrl REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_executable(deadzone_measure_node src/deadzone_measure_node.cpp)
ament_target_dependencies(deadzone_measure_node rclcpp rcl_interfaces std_msgs kaiev26_msgs lateral_ctrl)
add_executable(steering_bench_test_node src/steering_bench_test_node.cpp)
target_include_directories(steering_bench_test_node PRIVATE include)
ament_target_dependencies(steering_bench_test_node rclcpp rcl_interfaces std_msgs kaiev26_msgs)
install(TARGETS deadzone_measure_node steering_bench_test_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/mcap_to_csv.py scripts/torque_sysid_metrics.py
  scripts/record_mcap scripts/longitudinal_analysis DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE)
install(FILES README.md LOGGING.md DEADZONE_MEASUREMENT.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_bench_defaults test/test_bench_defaults.cpp)
  target_include_directories(test_bench_defaults PRIVATE include)
  ament_add_pytest_test(test_recording test TIMEOUT 120)
endif()
ament_package()
