cmake_minimum_required(VERSION 3.16)
project(path_mpc)
find_package(ament_cmake REQUIRED)
find_package(mpc_common REQUIRED)
find_package(rclcpp REQUIRED)
find_package(kaiev26_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(path_controller src/path_node.cpp)
target_compile_features(path_controller PUBLIC cxx_std_17)
target_link_libraries(path_controller mpc_common::mpc_solver Eigen3::Eigen yaml-cpp)
ament_target_dependencies(path_controller mpc_common rclcpp kaiev26_msgs ackermann_msgs sensor_msgs diagnostic_msgs geometry_msgs visualization_msgs)
install(TARGETS path_controller RUNTIME DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/fit_lateral_response scripts/evaluate_path DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
