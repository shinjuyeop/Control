cmake_minimum_required(VERSION 3.16)
project(steering_mpc)
find_package(ament_cmake REQUIRED)
find_package(mpc_common REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(steering_mpc_core SHARED src/steering_mpc.cpp)
target_compile_features(steering_mpc_core PUBLIC cxx_std_17)
target_include_directories(steering_mpc_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(steering_mpc_core mpc_common)
target_link_libraries(steering_mpc_core mpc_common::mpc_solver Eigen3::Eigen yaml-cpp)
install(TARGETS steering_mpc_core EXPORT steering_mpcTargets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(PROGRAMS scripts/fit_lateral scripts/evaluate_lateral DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/lateral_data.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(steering_mpcTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(mpc_common Eigen3 yaml-cpp)
ament_package()
